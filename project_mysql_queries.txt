CREATE DATABASE IF NOT EXISTS SmartTrafficAnalytics;
USE SmartTrafficAnalytics;

-- PART 1: DATABASE & TABLE CREATION (7 tables; no window functions)
CREATE TABLE Locations (
 location_id INT AUTO_INCREMENT PRIMARY KEY,
 location_name VARCHAR(120) NOT NULL, city VARCHAR(100) NOT NULL,
 address VARCHAR(255), area_type VARCHAR(50), description VARCHAR(255)
);

CREATE TABLE Roads (
 road_id INT AUTO_INCREMENT PRIMARY KEY, road_name VARCHAR(150) NOT NULL,
 road_type VARCHAR(50), speed_limit SMALLINT UNSIGNED, length_km DECIMAL(7,2),
 location_id INT NOT NULL,
 FOREIGN KEY (location_id) REFERENCES Locations(location_id)
);

CREATE TABLE Weather (
 weather_id INT AUTO_INCREMENT PRIMARY KEY, observation_date DATE NOT NULL,
 observation_time TIME NOT NULL, weather_condition VARCHAR(50) NOT NULL,
 temperature DECIMAL(5,2), visibility_km DECIMAL(6,2), precipitation_mm DECIMAL(7,2)
);

CREATE TABLE Vehicles (
 vehicle_id INT AUTO_INCREMENT PRIMARY KEY, vehicle_number VARCHAR(20) NOT NULL UNIQUE,
 vehicle_type VARCHAR(50) NOT NULL, manufacturer VARCHAR(80), model VARCHAR(80),
 manufacture_year YEAR
);

CREATE TABLE Traffic_Records (
 traffic_id INT AUTO_INCREMENT PRIMARY KEY, location_id INT NOT NULL, road_id INT NOT NULL,
 weather_id INT, observation_date DATE NOT NULL, observation_time TIME NOT NULL,
 vehicle_count INT UNSIGNED NOT NULL DEFAULT 0, avg_speed DECIMAL(6,2),
 traffic_density DECIMAL(5,2),
 FOREIGN KEY (location_id) REFERENCES Locations(location_id),
 FOREIGN KEY (road_id) REFERENCES Roads(road_id),
 FOREIGN KEY (weather_id) REFERENCES Weather(weather_id)
);

CREATE TABLE Accidents (
 accident_id INT AUTO_INCREMENT PRIMARY KEY, location_id INT NOT NULL, road_id INT NOT NULL,
 vehicle_id INT, weather_id INT, accident_date DATE NOT NULL, accident_time TIME NOT NULL,
 severity VARCHAR(30) NOT NULL, injuries INT UNSIGNED NOT NULL DEFAULT 0,
 fatalities INT UNSIGNED NOT NULL DEFAULT 0, cause VARCHAR(255), description TEXT,
 FOREIGN KEY (location_id) REFERENCES Locations(location_id),
 FOREIGN KEY (road_id) REFERENCES Roads(road_id),
 FOREIGN KEY (vehicle_id) REFERENCES Vehicles(vehicle_id),
 FOREIGN KEY (weather_id) REFERENCES Weather(weather_id)
);

CREATE TABLE Traffic_Violations (
 violation_id INT AUTO_INCREMENT PRIMARY KEY, vehicle_id INT NOT NULL, location_id INT NOT NULL,
 violation_type VARCHAR(100) NOT NULL, fine_amount DECIMAL(10,2) NOT NULL,
 violation_date DATE NOT NULL,
 payment_status ENUM('Paid','Pending','Overdue') NOT NULL DEFAULT 'Pending',
 description VARCHAR(255), CHECK (fine_amount >= 0),
 FOREIGN KEY (vehicle_id) REFERENCES Vehicles(vehicle_id),
 FOREIGN KEY (location_id) REFERENCES Locations(location_id)
);

-- PART 4: VIEWS
CREATE OR REPLACE VIEW view_traffic_overview AS
SELECT tr.traffic_id,tr.observation_date,tr.observation_time,l.location_name,l.city,
 r.road_name,tr.vehicle_count,tr.avg_speed,tr.traffic_density,w.weather_condition
FROM Traffic_Records tr JOIN Locations l ON tr.location_id=l.location_id
JOIN Roads r ON tr.road_id=r.road_id LEFT JOIN Weather w ON tr.weather_id=w.weather_id;

CREATE OR REPLACE VIEW view_accident_details AS
SELECT a.accident_id,a.accident_date,a.accident_time,l.location_name,l.city,r.road_name,
 v.vehicle_number,a.severity,a.injuries,a.fatalities,a.cause,w.weather_condition
FROM Accidents a JOIN Locations l ON a.location_id=l.location_id
JOIN Roads r ON a.road_id=r.road_id LEFT JOIN Vehicles v ON a.vehicle_id=v.vehicle_id
LEFT JOIN Weather w ON a.weather_id=w.weather_id;

CREATE OR REPLACE VIEW view_road_traffic_summary AS
SELECT r.road_id,r.road_name,COUNT(tr.traffic_id) AS observations,
 ROUND(AVG(tr.traffic_density),2) AS avg_density,ROUND(AVG(tr.avg_speed),2) AS avg_speed,
 SUM(tr.vehicle_count) AS total_observed_vehicles
FROM Roads r LEFT JOIN Traffic_Records tr ON r.road_id=tr.road_id
GROUP BY r.road_id,r.road_name;

CREATE OR REPLACE VIEW view_violation_summary AS
SELECT violation_type,COUNT(*) AS total_violations,SUM(fine_amount) AS total_fines,
 SUM(CASE WHEN payment_status='Paid' THEN fine_amount ELSE 0 END) AS paid_fines,
 SUM(CASE WHEN payment_status IN ('Pending','Overdue') THEN fine_amount ELSE 0 END) AS unpaid_fines
FROM Traffic_Violations GROUP BY violation_type;

-- PART 5: STORED PROCEDURES
DROP PROCEDURE IF EXISTS sp_accidents_by_location;
DELIMITER $$
CREATE PROCEDURE sp_accidents_by_location(IN p_location_id INT)
BEGIN
 SELECT accident_id,accident_date,accident_time,severity,injuries,fatalities,cause
 FROM Accidents WHERE location_id=p_location_id ORDER BY accident_date DESC,accident_time DESC;
END $$
DELIMITER ;

DROP PROCEDURE IF EXISTS sp_road_traffic_summary;
DELIMITER $$
CREATE PROCEDURE sp_road_traffic_summary(IN p_road_id INT)
BEGIN
 SELECT r.road_name,COUNT(tr.traffic_id) AS observations,
 ROUND(AVG(tr.vehicle_count),2) AS avg_vehicle_count,
 ROUND(AVG(tr.avg_speed),2) AS avg_speed,ROUND(AVG(tr.traffic_density),2) AS avg_density
 FROM Roads r LEFT JOIN Traffic_Records tr ON r.road_id=tr.road_id
 WHERE r.road_id=p_road_id GROUP BY r.road_id,r.road_name;
END $$
DELIMITER ;

DROP PROCEDURE IF EXISTS sp_add_violation;
DELIMITER $$
CREATE PROCEDURE sp_add_violation(IN p_vehicle_id INT,IN p_location_id INT,
 IN p_violation_type VARCHAR(100),IN p_fine_amount DECIMAL(10,2),
 IN p_violation_date DATE,IN p_description VARCHAR(255))
BEGIN
 IF p_fine_amount<0 THEN
  SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Fine amount cannot be negative';
 ELSE
  INSERT INTO Traffic_Violations(vehicle_id,location_id,violation_type,fine_amount,violation_date,description)
  VALUES(p_vehicle_id,p_location_id,p_violation_type,p_fine_amount,p_violation_date,p_description);
 END IF;
END $$
DELIMITER ;

-- PART 6: TRIGGERS
CREATE TABLE IF NOT EXISTS violation_audit (
 audit_id INT AUTO_INCREMENT PRIMARY KEY,violation_id INT NOT NULL,
 action_type VARCHAR(10) NOT NULL,old_fine DECIMAL(10,2),new_fine DECIMAL(10,2),
 changed_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
);

DROP TRIGGER IF EXISTS trg_validate_violation_insert;
DELIMITER $$
CREATE TRIGGER trg_validate_violation_insert BEFORE INSERT ON Traffic_Violations
FOR EACH ROW
BEGIN
 IF NEW.fine_amount<0 THEN SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Fine amount cannot be negative'; END IF;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_audit_violation_update;
DELIMITER $$
CREATE TRIGGER trg_audit_violation_update AFTER UPDATE ON Traffic_Violations
FOR EACH ROW
BEGIN
 IF OLD.fine_amount<>NEW.fine_amount THEN
  INSERT INTO violation_audit(violation_id,action_type,old_fine,new_fine)
  VALUES(NEW.violation_id,'UPDATE',OLD.fine_amount,NEW.fine_amount);
 END IF;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_audit_violation_delete;
DELIMITER $$
CREATE TRIGGER trg_audit_violation_delete AFTER DELETE ON Traffic_Violations
FOR EACH ROW
BEGIN
 INSERT INTO violation_audit(violation_id,action_type,old_fine,new_fine)
 VALUES(OLD.violation_id,'DELETE',OLD.fine_amount,NULL);
END $$
DELIMITER ;
